```rust
//! The single width definition used throughout envision.

use ratatui::buffer::CellWidth;

/// Display width of `symbol` in terminal cells, as ratatui computes it.
///
/// A single ASCII control byte (reachable via `set_char`/`set_symbol`) is
/// reported as 1 — the value ratatui's release builds return — without calling
/// `CellWidth`, whose `debug_assert!` panics on such bytes in debug builds.
pub(crate) fn symbol_cell_width(symbol: &str) -> u16 {
    if symbol.len() == 1 && symbol.as_bytes()[0].is_ascii_control() {
        1
    } else {
        symbol.cell_width()
    }
}

#[cfg(test)]
mod tests {

    #[test]

        assert_eq!(symbol_cell_width("a"), 1);

    #[test]
    fn cjk_is_two() {
        assert_eq!(symbol_cell_width("中"), 2);
    }

    #[test]
    fn empty_is_zero() {
        assert_eq!(symbol_cell_width(""), 0);
    }

    #[test]
```

```rust
//! Serializable mirror of ratatui's `CellDiffOption`.

use std::num::NonZeroU16;

use ratatui::buffer::CellDiffOption;

/// How a cell participates in buffer diffing, captured from ratatui.
///
/// `#[non_exhaustive]` — unlike [`SerializableColor`](super::SerializableColor) —
/// because ratatui's `CellDiffOption` has already grown variants and hyperlink
/// support (#125) will exercise them; the mirror keeps room to follow without
/// a breaking change.
#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
#[cfg_attr(feature = "serialization", derive(serde::Serialize, serde::Deserialize))]
#[cfg_attr(feature = "serialization", serde(rename_all = "snake_case"))]
#[non_exhaustive]
pub enum SerializableDiffOption {
    /// No special handling.
    #[default]
    None,
    /// Skipped when diffing.
    Skip,
    /// Always re-sent, even if unchanged.
    AlwaysUpdate,
    /// Occupies exactly this many columns regardless of the symbol.
    ForcedWidth(NonZeroU16),
}

impl SerializableDiffOption {
    /// True for [`SerializableDiffOption::None`].
    pub fn is_none(&self) -> bool {
        matches!(self, Self::None)
    }
}

impl From<CellDiffOption> for SerializableDiffOption {
    fn from(option: CellDiffOption) -> Self {
        match option {
            CellDiffOption::None => Self::None,
            CellDiffOption::Skip => Self::Skip,
            CellDiffOption::AlwaysUpdate => Self::AlwaysUpdate,
            CellDiffOption::ForcedWidth(w) => Self::ForcedWidth(w),
        }
    }
}
```

```rust
//! Terminal wide-glyph semantics for captured cells (spec § 5).
//!
//! Widths come from `EnhancedCell::covered_width` (ratatui's width via the
//! crate's single helper). No per-cell ownership state is stored: the owner of
//! a column is found by scanning left over blank cells.

use super::CaptureBackend;
use crate::backend::cell::EnhancedCell;

impl CaptureBackend {
    /// Writes `cell` at `(x, y)` (in bounds), applying spec § 5 rules 1–4.
    pub(super) fn write_cell(&mut self, x: u16, y: u16, mut cell: EnhancedCell) {
        let frame = self.current_frame;

        // Rules 2 and 3: a write into a continuation column.
        if !cell.is_blank_symbol() {
            if let Some(owner_x) = self.owner_of(x, y) {
                self.blank_owner_and_continuations(owner_x, y, frame); // rule 3
            }
        }
        // (A blank write into a continuation column — rule 2 — is just a write:
        // the owner stays intact and the written style is recorded.)

        // Rule 4: replacing an owner blanks its former continuation cells.
        let idx = self.index_of(x, y);
        let old_width = self.cells[idx].covered_width();
        if old_width > 1 {
            self.blank_range(x.saturating_add(1), x.saturating_add(old_width), y, frame);
        }

        let new_width = cell.covered_width();
        cell.set_last_modified_frame(frame);
        self.cells[idx] = cell;

        // Rule 1: a new owner's continuation cells become reset blanks.
        if new_width > 1 {
            let end = x.saturating_add(new_width).min(self.width);
            for cx in x.saturating_add(1)..end {
                let mut blank = EnhancedCell::new();
                blank.set_last_modified_frame(frame);
                let cidx = self.index_of(cx, y);
                self.cells[cidx] = blank;
            }
        }
    }

    /// The column of the wide glyph covering `(x, y)` as a continuation, if any.
    /// Scans left only across blank cells, so the cost is bounded by the blank
    /// run immediately to the left.
    fn owner_of(&self, x: u16, y: u16) -> Option<u16> {
        let mut candidate = x;
        while candidate > 0 {
            candidate -= 1;
            let c = &self.cells[self.index_of(candidate, y)];
            if !c.is_blank_symbol() {
                let reach = candidate.saturating_add(c.covered_width());
                return (reach > x).then_some(candidate);
            }
        }
        None
    }

    /// Rule 3: blank the owner and its continuation cells, keeping their style.
    fn blank_owner_and_continuations(&mut self, owner_x: u16, y: u16, frame: u64) {
        let width = self.cells[self.index_of(owner_x, y)].covered_width();
        self.blank_range(owner_x, owner_x.saturating_add(width), y, frame);
    }

    /// Sets the symbol of in-row cells `[from, to)` to `" "`, keeping style.
    fn blank_range(&mut self, from: u16, to: u16, y: u16, frame: u64) {
        for cx in from..to.min(self.width) {
            let idx = self.index_of(cx, y);
            self.cells[idx].set_symbol(" ");
            self.cells[idx].set_last_modified_frame(frame);
        }
    }
}
```

```python
"""Four-part proof that each .snap change is upstream canvas rasterization (spec § 6)."""
import pathlib, sys

BRAILLE = range(0x2800, 0x2900)

def grid(path):
    text = pathlib.Path(path).read_text()
    body = text.split("---", 2)[2].strip("\n")      # drop insta header
    return [list(line) for line in body.split("\n")]

def is_dot_or_space(ch):
    return ch == " " or ord(ch) in BRAILLE

def dots(g):
    pts = set()
    for y, row in enumerate(g):
        for x, ch in enumerate(row):
            if ord(ch) in BRAILLE:
                bits = ord(ch) - 0x2800
                for bit, (dx, dy) in enumerate([(0,0),(0,1),(0,2),(1,0),(1,1),(1,2),(0,3),(1,3)]):
                    if bits & (1 << bit):
                        pts.add((x * 2 + dx, y * 4 + dy))
    return pts

def nearest(p, pts):
    return min((max(abs(p[0]-q[0]), abs(p[1]-q[1])) for q in pts), default=99)

def check(old_path, new_path, label):
    old, new = grid(old_path), grid(new_path)
    assert len(old) == len(new) and all(len(a) == len(b) for a, b in zip(old, new)), "1: dimensions differ"
    changed = [(x, y) for y, (a, b) in enumerate(zip(old, new)) for x, (c, d) in enumerate(zip(a, b)) if c != d]
    for x, y in changed:
        assert is_dot_or_space(old[y][x]) and is_dot_or_space(new[y][x]), f"2: non-plot cell changed at {x},{y}"
    xs = [x for x, _ in changed]; ys = [y for _, y in changed]
    print(f"   3: changed cells span x={min(xs)}..{max(xs)} y={min(ys)}..{max(ys)} (verify inside graph area)")
    od, nd = dots(old), dots(new)
    if label == "rounding":
        h = max(max(nearest(p, nd) for p in od), max(nearest(p, od) for p in nd))
        assert h <= 1, f"4: Hausdorff {h} > 1"
    else:
        assert all(nearest(p, nd) <= 1 for p in od), "4: an old dot has no new dot within 1"
        extra = sorted(p for p in nd if nearest(p, od) > 1)
        print(f"   4: {len(extra)} extra dots from clipping: {extra[:12]}…")
    print("   OK")

if __name__ == "__main__":
    for line in pathlib.Path(sys.argv[1]).read_text().split("\n"):
        if line.strip():
            new_path, label = line.split()
            print(new_path, label)
            check(new_path[: -len(".new")], new_path, label)
```
